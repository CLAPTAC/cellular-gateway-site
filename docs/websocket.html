Devices connect to:

WS /ws/devices?device_key=...&device_token=...

  "type": "command",
  "command": {
    "type": "SMS_SEND",
    "request_id": "req_...",
    "device_id": "...",
    "payload": { "to": "+94...", "message": "Hello" }
  }
}
